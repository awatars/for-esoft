Задание 2: Глубокое копирование объекта

Функция deepClone выполняет глубокое копирование объекта с поддержкой вложенных структур, циклических ссылок, Date, Map, Set, RegExp, функций, символов и прототипов.

JavaScript 
/**
 * Глубокое копирование объекта.
 *
 * Поддерживает: Object, Array, Date, RegExp, Map, Set,
 * функции, символьные ключи, циклические ссылки, прототипы.
 *
 * @param {*} source  исходное значение
 * @param {WeakMap} [seen]  карта уже скопированных объектов (для циклов)
 * @returns {*} глубокая копия
 */
function deepClone(source, seen = new WeakMap()) {

    // 1. Примитивы и null: возвращаем как есть
    if (source === null || typeof source !== 'object' && typeof source !== 'function') {
        return source;
    }

    // 2. Функции: копируем, привязывая к тому же контексту
    if (typeof source === 'function') {
        const clonedFn = source.bind({});
        Object.defineProperties(clonedFn,
            Object.getOwnPropertyDescriptors(source)
        );
        return clonedFn;
    }

    // 3. Проверка циклических ссылок
    if (seen.has(source)) {
        return seen.get(source);
    }

    // 4. Date
    if (source instanceof Date) {
        return new Date(source.getTime());
    }

    // 5. RegExp
    if (source instanceof RegExp) {
        return new RegExp(source.source, source.flags);
    }

    // 6. Map
    if (source instanceof Map) {
        const mapCopy = new Map();
        seen.set(source, mapCopy);
        source.forEach((val, key) => {
            mapCopy.set(deepClone(key, seen), deepClone(val, seen));
        });
        return mapCopy;
    }

    // 7. Set
    if (source instanceof Set) {
        const setCopy = new Set();
        seen.set(source, setCopy);
        source.forEach((val) => {
            setCopy.add(deepClone(val, seen));
        });
        return setCopy;
    }

    // 8. Array и Object: сохраняем прототип
    const proto = Object.getPrototypeOf(source);
    const clone = Array.isArray(source)
        ? []
        : Object.create(proto);

    seen.set(source, clone);

    // 9. Копируем все собственные свойства (включая символьные)
    const keys = [
        ...Object.getOwnPropertyNames(source),
        ...Object.getOwnPropertySymbols(source)
    ];

    for (const key of keys) {
        const descriptor = Object.getOwnPropertyDescriptor(source, key);
        if (descriptor.value !== undefined) {
            descriptor.value = deepClone(descriptor.value, seen);
        }
        Object.defineProperty(clone, key, descriptor);
    }

    return clone;
}
